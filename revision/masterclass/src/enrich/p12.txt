G js:
// minimal circuit breaker: closed → open after N failures → half-open probe after cooldown
class CircuitBreaker {
  constructor({ failures = 5, cooldownMs = 30_000 } = {}) { Object.assign(this, { failures, cooldownMs, count: 0, openedAt: 0 }); }
  get state() { return this.count < this.failures ? 'closed' : Date.now() - this.openedAt > this.cooldownMs ? 'half-open' : 'open'; }
  async call(fn) {
    if (this.state === 'open') throw new Error('circuit open');
    try { const r = await fn(); this.count = 0; return r; }
    catch (e) { if (++this.count >= this.failures) this.openedAt = Date.now(); throw e; }
  }
}
