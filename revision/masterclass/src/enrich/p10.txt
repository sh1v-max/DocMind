  const msgs = await repo.recentMessages(conversationId, userId, 30);          // newest first, scoped to the user
  const out = []; let used = 0;

async function assertPublicUrl(raw) {
  const url = new URL(raw);
  if (!['http:', 'https:'].includes(url.protocol)) throw new HttpError(400, 'Only http(s) URLs');
  const addrs = net.isIP(url.hostname) ? [url.hostname] : (await dns.lookup(url.hostname, { all: true })).map((a) => a.address);
  if (addrs.some(isPrivate)) throw new HttpError(400, 'URL not allowed');
  return url;   // also: fetch with redirect: 'manual' and re-check each hop; cap size and time
}
